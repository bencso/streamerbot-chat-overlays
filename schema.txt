 client.on("Twitch.Sub", (event) => {
    const data = event.data || event;
    const structuredData = {
      type: event.event.type,
      userId: data.user.id,
      displayName: data.user.name,
      tier: data.sub_tier,
      duration: data.duration_months,
    };
    addQueueItem(
      "alert",
      structuredData,
      data.user,
      event.source.toLowerCase(),
    );
  });
